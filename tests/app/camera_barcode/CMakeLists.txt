cmake_minimum_required(VERSION 3.20)
project(camera_barcode_test C)
set(ZXC_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../modules/lib/zxc" CACHE PATH "ZXC checkout")
set(ZXC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZXC_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
option(ZXC_USE_HEAP "Test heap1 instead of heap0" OFF)
set(ALLOCATOR_TEST_DIR "${CMAKE_CURRENT_LIST_DIR}/../../lib/zxc_allocator")
if(ZXC_USE_HEAP)
  set(ZXC_ALLOCATOR_BACKEND CUSTOM CACHE STRING "" FORCE)
  set(ZXC_ALLOCATOR_CUSTOM_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../app/src/scanner_zxc.c" CACHE FILEPATH "" FORCE)
endif()
add_subdirectory("${ZXC_DIR}" zxc)
if(ZXC_USE_HEAP)
  target_include_directories(zxc PRIVATE "${ALLOCATOR_TEST_DIR}/include")
  target_compile_definitions(zxc PRIVATE
    CONFIG_ZXC_USE_HEAP=1 CONFIG_ZXC_ALLOCATOR_CUSTOM=1
    CONFIG_VIDEO_BUFFER_USE_SHARED_MULTI_HEAP=1 CONFIG_VIDEO_BUFFER_SMH_ATTRIBUTE=2)
  target_sources(zxc PRIVATE "${ALLOCATOR_TEST_DIR}/shared_heap_stub.c")
  # Keep the shared heap stub's checks enabled in Release builds too.
  set_source_files_properties("${ALLOCATOR_TEST_DIR}/shared_heap_stub.c"
    TARGET_DIRECTORY zxc PROPERTIES COMPILE_OPTIONS -UNDEBUG)
endif()
add_executable(camera_barcode_test main.c)
if(NOT ZXC_USE_HEAP)
  target_sources(camera_barcode_test PRIVATE ../../../app/src/scanner_zxc.c)
endif()
target_include_directories(camera_barcode_test PRIVATE ../../../app/src "${ZXC_DIR}/samples/zephyr/scan/src")
if(ZXC_USE_HEAP)
  target_include_directories(camera_barcode_test PRIVATE "${ALLOCATOR_TEST_DIR}/include")
endif()
target_link_libraries(camera_barcode_test PRIVATE zxc)
target_compile_options(camera_barcode_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME camera_barcode COMMAND camera_barcode_test)
