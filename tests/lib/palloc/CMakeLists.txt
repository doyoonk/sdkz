cmake_minimum_required(VERSION 3.20)
project(palloc_tests C)
find_package(Threads REQUIRED)
option(PALLOC_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" ON)
add_executable(palloc_test main.c)
target_compile_definitions(palloc_test PRIVATE CONFIG_HU_PALLOC=1)
add_executable(palloc_fallback_test fallback.c ../../../lib/hu/palloc.c)
target_compile_definitions(palloc_fallback_test PRIVATE CONFIG_HU_PALLOC=0)
add_executable(palloc_no_dtcm_test main.c)
target_compile_definitions(palloc_no_dtcm_test PRIVATE CONFIG_HU_PALLOC=1 TEST_HAS_DTCM=0)
foreach(target palloc_test palloc_no_dtcm_test)
  target_sources(${target} PRIVATE ../../../app/src/heap_test.c)
endforeach()
enable_testing()
foreach(target palloc_test palloc_fallback_test palloc_no_dtcm_test)
  target_compile_features(${target} PRIVATE c_std_11)
  target_include_directories(${target} PRIVATE include ../../../include)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(PALLOC_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
